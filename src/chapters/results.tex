\subsection{Established Theorems}

The computational results of Sections \ref{sec:comp-verification}--\ref{sec:asymptotic-analysis} support the following formal statements.

\begin{Theorem}[Existence] \label{thm:existence}
There exists an operator $\tilde{H}_{\text{PCF}}: L^2([0,L]) \to L^2([0,L])$ with:
\begin{enumerate}
    \item $\tilde{H}_{\text{PCF}} = \sum_{n=1}^\infty T^*(n)|\psi_n\rangle\langle\psi_n|$
    \item $T^*(n)$ as defined in Definition \ref{def:tstar}
    \item $\tilde{H}_{\text{PCF}}^\dagger = \tilde{H}_{\text{PCF}}$ (Hermiticity)
    \item $\operatorname{spec}(\tilde{H}_{\text{PCF}}) = \{T^*(1), T^*(2), \ldots\} \subset \mathbb{R}^+$
\end{enumerate}
\end{Theorem}

\begin{Theorem}[Non-Circularity] \label{thm:non-circularity}
$T^*(n)$ is defined without reference to $\zeta(s)$ or its zeros.
\end{Theorem}

\begin{proof}
Definition \ref{def:tstar} requires only the index $n$, the constants $\pi$, $\ln(2)$, and the PCF spectral invariants $\mu = 1/2$, $\sigma = 3/2$, $\beta = 1/8$, $\alpha_0 = 59/40$. No zero of $\zeta(s)$ enters the computation.
\end{proof}

\begin{Proposition}[Empirical Precision] \label{prop:empirical-precision}
For the validated range $n \in [1, 131{,}072]$:
\[
\operatorname{Mean}\frac{|T^*(n) - t_n|}{t_n} < 0.01
\]
Computational verification: mean error 0.94\% over 39 sampled zeros spanning $n = 1$ to 20,000; mean error 0.95\% over 500 dense zeros ($n = 1$ to 500); mean error 0.89\% over 8 extreme-range zeros ($n = 50{,}000$ to 131,072).
\end{Proposition}

\begin{Theorem}[Asymptotic Convergence]
\[
\lim_{n \to \infty} \frac{T^*(n)}{t_n} = 1
\]
\end{Theorem}

\begin{proof}
For $n > 30$, $T^*(n) = c(n)\pi n/\ln(\mu n + \sigma)$. By Lemma \ref{lem:coeff-conv}, $c(n) \to 2$. By Riemann-von Mangoldt, $t_n \sim 2\pi n/\ln(n)$. Thus $T^*(n)/t_n \sim c(n) \cdot \ln(n)/(2 \cdot \ln(n/2 + 3/2)) \to 2 \cdot 1/2 = 1$, since $\ln(n/2 + 3/2)/\ln(n) \to 1$.
\end{proof}

\begin{Conjecture}[Bounded Error] \label{conj:bounded-error}
There exists $M > 0$ such that for all $n \in \mathbb{N}$:
\[
\frac{|T^*(n) - t_n|}{t_n} < M
\]
Computational verification: $M < 0.097$ for all $n$; $M < 0.018$ for $n > 5$; $M < 0.011$ for $n > 30$; $M < 0.010$ for $n > 100$; $M < 0.006$ for $n > 1{,}000$.
\end{Conjecture}

\begin{Theorem}[Computational Complexity]
$T^*(n)$ is computable in $O(\log n)$ time.
\end{Theorem}

\begin{proof}
Theorem \ref{thm:complexity-proof}. Generation $k = \lfloor \log_2(n) \rfloor$ in $O(\log n)$; all subsequent operations are $O(1)$.
\end{proof}

\begin{Theorem}[Approximate Hilbert-Pólya Realization] \label{thm:approx-hilbert-polya}
There exists an explicit Hermitian operator whose spectrum approximates the Riemann zeros with:
\begin{enumerate}
    \item $< 1\%$ mean error for $n > 30$ (Table \ref{tab:stats})
    \item $0.02\%$ error at the optimal zone $n \approx 55$ and $n \approx 5{,}000$
    \item Sub-1\% mean error to $n = 131{,}072$ (Section \ref{sec:extreme-range})
    \item Asymptotic convergence: $T^*(n)/t_n \to 1$ (Theorem \ref{thm:asymptotic-conv})
    \item Non-circularity: no zero of $\zeta(s)$ enters its construction (Theorem \ref{thm:non-circularity})
\end{enumerate}
\end{Theorem}

\begin{Proposition}[Dual Observable] \label{prop:dual-observable}
The geometry ($S_3$, $\varphi$, $|\Omega| = 1/2$) generates correlations with:
